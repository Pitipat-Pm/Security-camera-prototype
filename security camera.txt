from flask import Flask, Response, render_template_string, request, send_from_directory, jsonify
from picamera2 import Picamera2
from picamera2.encoders import H264Encoder
import cv2, threading, time, os, psutil, csv

# ---------------- CONFIG ----------------
CAM_RESOLUTION = (320, 240)
FPS = 8

photo_dir = "photos"
video_dir = "videos"
record_dir = "recordings"
os.makedirs(photo_dir, exist_ok=True)
os.makedirs(video_dir, exist_ok=True)
os.makedirs(record_dir, exist_ok=True)

face_cascade = cv2.CascadeClassifier("haarcascade_frontalface_default.xml")
csv_file = os.path.join(record_dir, "object_log.csv")
if not os.path.exists(csv_file):
    with open(csv_file, 'w', newline='') as f:
        csv.writer(f).writerow(['timestamp','faces_detected','total_detect_count'])

# ---------------- CAMERA ----------------
picam2 = Picamera2()
config = picam2.create_preview_configuration(main={"size": CAM_RESOLUTION})
picam2.configure(config)
picam2.start()
time.sleep(1)

frame_buffer = None
latest_photo = None
running = False
camera_thread = None
mode = None
total_detect_count = 0
recording = False
h264_file = None
encoder = None

# ---------------- CAMERA LOOP ----------------
def camera_loop():
    global frame_buffer, running, mode, total_detect_count, latest_photo
    last_detect_time = time.time()
    while running:
        frame_start_time = time.time()
        frame = picam2.capture_array()

        # CPU / Temp
        cpu = psutil.cpu_percent()
        try:
            with open("/sys/class/thermal/thermal_zone0/temp") as f:
                temp = float(f.read())/1000
        except:
            temp = 0.0

        faces_detected = 0

        # PHOTO
        if mode == "photo":
            timestamp = time.strftime("%Y-%m-%d_%H-%M-%S")
            filename = f"{photo_dir}/{timestamp}.jpg"
            cv2.imwrite(filename, frame)
            latest_photo = filename
            mode = None

        # DETECTION
        elif mode == "detection":
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            faces = face_cascade.detectMultiScale(gray, scaleFactor=1.2, minNeighbors=5)
            faces_detected = len(faces)
            for (x, y, w, h) in faces:
                cv2.rectangle(frame, (x, y, w, h), (0, 255, 0), 2)

            now = time.time()
            if now - last_detect_time >= 1.0:  # ตรวจจับทุก 1 วินาที
                total_detect_count += faces_detected
                with open(csv_file, 'a', newline='') as f:
                    csv.writer(f).writerow([time.strftime('%Y-%m-%d %H:%M:%S'), faces_detected, total_detect_count])
                last_detect_time = now

            print(f"Mode: {mode} | Faces now: {faces_detected} | Total detects: {total_detect_count} | CPU: {cpu}% | Temp: {temp:.1f}°C | FPS: {FPS}")

        # VIDEO
        elif mode == "video":
            print(f"Mode: {mode} | CPU: {cpu}% | Temp: {temp:.1f}°C | FPS: {FPS}")

        # FRAME BUFFER
        ret, buffer = cv2.imencode('.jpg', frame)
        if ret:
            frame_buffer = buffer.tobytes()

        frame_time = time.time() - frame_start_time
        time.sleep(max(0, (1/FPS) - frame_time))

# ---------------- VIDEO LOOP ----------------
def video_loop():
    global recording, h264_file, encoder
    timestamp = time.strftime("%Y-%m-%d_%H-%M-%S")
    h264_file = f"{video_dir}/{timestamp}.h264"
    encoder = H264Encoder(bitrate=1000000)
    picam2.start_recording(encoder, h264_file)
    print("Recording started...")
    while recording:
        time.sleep(0.1)

@app.route('/latest_photo')
def latest_photo_api():
    global latest_photo
    return jsonify({"path": latest_photo})

@app.route('/view_file')
def view_file_param():
    path = request.args.get('path')
    if path and os.path.exists(path):
        ext = path.split('.')[-1].lower()
        if ext in ['jpg','jpeg','png']:
            mimetype='image/'+ext
        elif ext in ['mp4']:
            mimetype='video/mp4'
        else:
            mimetype='application/octet-stream'
        return send_from_directory(os.path.dirname(path), os.path.basename(path), mimetype=mimetype)
    return "File not found", 404

@app.route('/file_lists')
def file_lists():
    photos = sorted(os.listdir(photo_dir), reverse=True)
    videos = sorted([f for f in os.listdir(video_dir) if f.endswith('.mp4')], reverse=True)
    recordings = sorted(os.listdir(record_dir), reverse=True)
    html = ""

    html += "<h2>Photos</h2>"
    for f in photos:
        html += f'<a href="/view/photos/{f}" target="_blank">{f}</a> ' \
                f'<form action="/delete/photos/{f}" method="post" style="display:inline;">' \
                f'<button type="submit">Delete</button></form><br>'

    html += "<h2>Videos</h2>"
    for f in videos:
        html += f'<video width="320" controls><source src="/view/videos/{f}" type="video/mp4"></video> ' \
                f'<form action="/delete/videos/{f}" method="post" style="display:inline;">' \
                f'<button type="submit">Delete</button></form><br>'

    html += "<h2>Recordings / CSV</h2>"
    for f in recordings:
        html += f'<a href="/view/recordings/{f}" target="_blank">{f}</a> ' \
                f'<form action="/delete/recordings/{f}" method="post" style="display:inline;">' \
                f'<button type="submit">Delete</button></form><br>'
    return html

@app.route('/video_feed')
def video_feed():
    def generate_mjpeg():
        global frame_buffer
        while True:
            if frame_buffer:
                yield (b'--frame\r\nContent-Type: image/jpeg\r\n\r\n'+frame_buffer+b'\r\n')
            else:
                time.sleep(0.05)
    return Response(generate_mjpeg(), mimetype='multipart/x-mixed-replace; boundary=frame')

@app.route('/start', methods=['POST'])
def start():
    global running, camera_thread, mode, recording
    m = request.form.get('mode')
    if m:
        mode = m

        # เริ่ม camera_loop ถ้ายังไม่รัน
        if not running:
            running = True
            camera_thread = threading.Thread(target=camera_loop, daemon=True)
            camera_thread.start()

        # ถ้า Video mode เริ่ม video_loop
        if mode == "video" and not recording:
            recording = True
            t_video = threading.Thread(target=video_loop, daemon=True)
            t_video.start()

    return ("", 204)

@app.route('/stop_save', methods=['POST'])
def stop_save():
    global mode, recording, h264_file
    response_msg = ""
    if mode == "video" and h264_file and os.path.exists(h264_file):
        recording = False
        picam2.stop_recording()
        mp4_file = h264_file.replace(".h264", ".mp4")
        os.system(f"MP4Box -add {h264_file} {mp4_file}")
        os.remove(h264_file)
        response_msg = f"Video saved: {mp4_file}"
        h264_file = None
    elif mode == "detection":
        response_msg = f"Detection stopped. CSV saved: {csv_file}"
    else:
        response_msg = f"{mode or 'All'} mode stopped."
    mode = None
    return jsonify({"message": response_msg})

@app.route('/reset', methods=['POST'])
def reset():
    global mode, recording, running, camera_thread
    mode = None
    recording = False
    running = False
    time.sleep(0.1)
    running = True
    camera_thread = threading.Thread(target=camera_loop, daemon=True)
    camera_thread.start()
    return jsonify({"message": "System reset. Ready for next function."})

@app.route('/delete/<folder>/<filename>', methods=['POST'])
def delete_file(folder, filename):
    folders = {"photos": photo_dir, "videos": video_dir, "recordings": record_dir}
    if folder in folders:
        file_path = os.path.join(folders[folder], filename)
        if os.path.exists(file_path):
            os.remove(file_path)
            return ("", 204)
        else:
            return "File not found", 404
    return "Folder not found", 404

@app.route('/view/<folder>/<filename>')
def view_file_folder(folder, filename):
    folders = {"photos": photo_dir, "videos": video_dir, "recordings": record_dir}
    if folder in folders:
        path = os.path.join(folders[folder], filename)
        if os.path.exists(path):
            ext = filename.split('.')[-1].lower()
            if ext in ['jpg','jpeg','png']:
                mimetype='image/'+ext
            elif ext in ['mp4']:
                mimetype='video/mp4'
            else:
                mimetype='application/octet-stream'
            return send_from_directory(folders[folder], filename, mimetype=mimetype)
    return "File not found", 404
